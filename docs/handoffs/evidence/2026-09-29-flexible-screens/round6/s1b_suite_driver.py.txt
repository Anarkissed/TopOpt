#!/usr/bin/env python3
# S1b targeted suite driver: runs the class list in chunks with `swift test --skip-build` (unseeded, CI's mode),
# resumes a chunk from its first unfinished class after an xctest death, and skips a class that dies twice in a row
# (deterministic). Never kills anything. Writes suite.log (raw) and suite_summary.txt (per class) in this folder.
import os, re, subprocess, sys, time

S = os.path.dirname(os.path.abspath(__file__))
PKG = "/Users/nadim/dev/TopOpt/TopOpt/.claude/worktrees/flexible-screens-track-9be87b/app/TopOptKit"
CHUNK = 12
entries = []
for line in open(os.path.join(S, "suite_classes.txt")):
    line = line.split("#")[0].strip()
    if line:
        b, c = line.split(".", 1)
        entries.append((b, c))

log = open(os.path.join(S, "suite.log"), "a", buffering=1)
summary = open(os.path.join(S, "suite_summary.txt"), "a", buffering=1)
suite_re = re.compile(r"Test Suite '([A-Za-z0-9_]+)' (passed|failed) at")
exec_re = re.compile(r"^\s*Executed (\d+) tests?, with (\d+) tests? skipped and (\d+) failures?|^\s*Executed (\d+) tests?, with (\d+) failures?")

def run(bundle, classes):
    flt = "%s\\.(%s)/" % (bundle, "|".join(classes))
    log.write("\n### RUN %s %s %d classes: %s\n" % (time.strftime("%H:%M:%S"), bundle, len(classes), " ".join(classes)))
    p = subprocess.Popen(["swift", "test", "--skip-build", "--filter", flt], cwd=PKG,
                         stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, errors="replace")
    done = {}
    awaiting = None
    last_exec = {}
    for line in p.stdout:
        log.write(line)
        m = exec_re.search(line)
        if m and awaiting is not None:
            last_exec[awaiting] = line.strip()
            awaiting = None
        m = suite_re.search(line)
        if m and m.group(1) in classes:
            done[m.group(1)] = m.group(2)
            awaiting = m.group(1)
    rc = p.wait()
    log.write("### EXIT %d at %s\n" % (rc, time.strftime("%H:%M:%S")))
    return rc, done, last_exec

for b in sorted({b for b, _ in entries}):
    pending = [c for bb, c in entries if bb == b]
    died = {}
    while pending:
        chunk = pending[:CHUNK]
        rc, done, execs = run(b, chunk)
        for c in chunk:
            if c in done:
                summary.write("%s %s.%s %s\n" % (done[c].upper(), b, c, execs.get(c, "")))
        unfinished = [c for c in chunk if c not in done]
        if unfinished:
            first = unfinished[0]
            died[first] = died.get(first, 0) + 1
            if died[first] >= 2:
                summary.write("DIED-TWICE %s.%s (deterministic; skipped) rc %d\n" % (b, first, rc))
                unfinished = unfinished[1:]
            else:
                summary.write("DIED %s.%s rc %d — resuming from it\n" % (b, first, rc))
        pending = unfinished + pending[CHUNK:]
summary.write("SUITE DONE %s\n" % time.strftime("%H:%M:%S"))
